% Custom commands
\newcommand{\resumeItemSimple}[1]{
  \item\small{
    {#1 \vspace{-2pt}}
  }
}
\newcommand{\resumeItem}[2]{
  \item\small{
    \textbf{#1}{: #2 \vspace{-2pt}}
  }
}

\newcommand{\resumeSubheadingExtended}[6]{
  \vspace{-1pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} -- \textit{\small#2} & \small#3 \\
       \footnotesize#4; \footnotesize#5 & \textit{\small #6} \\
    \end{tabular*}\vspace{-5pt}
}

\newcommand{\resumeSubheading}[4]{
  \vspace{-1pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-5pt}
}

\newcommand{\resumeSubSubheading}[2]{
    \begin{tabular*}{0.97\textwidth}{l@{\extracolsep{\fill}}r}
      \textit{\small#1} & \textit{\small #2} \\
    \end{tabular*}\vspace{-5pt}
}

\newcommand{\resumeSubItem}[2]{\resumeItem{#1}{#2}\vspace{-4pt}}

\renewcommand{\labelitemii}{$\circ$}

\newcommand{\resumeSubHeadingListStart}{\begin{itemize}[leftmargin=*]}
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}}
\newcommand{\resumeItemListStart}{\begin{itemize}[leftmargin=*,topsep=0pt]}
\newcommand{\resumeItemListEnd}{\end{itemize}\vspace{-5pt}}

\definecolor{tag_background}{RGB}{234,234,234}
\definecolor{tag_text}{RGB}{38,65,109}

\newcommand\sotag[1]{%
    \tikz[baseline]{%
        \node[anchor=base, text=tag_text, fill=tag_background, text depth=.3mm, scale = 0.8] {#1};
    }%
}

    \resumeSubheading

      {\href{\VAR{e.link}}{\VAR{e.company}}}{\VAR{e.date}}
    \par\vspace{6pt}
    \resumeItemListStart
      \BLOCK{ for c in e.captions }
        \resumeItem{\VAR{c.name}}{\VAR{c.content | link_projects(project_links, c.highlights | default([]))}}
